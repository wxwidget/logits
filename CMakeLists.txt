cmake_minimum_required(VERSION 3.24)
project(LogitLab VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LOGITLAB_PYTHON "Build Python bindings" OFF)
option(LOGITLAB_LEGACY "Build historical research executables" OFF)
option(LOGITLAB_NATIVE "Optimize for the current CPU; not portable" OFF)
include(FetchContent)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(abseil GIT_REPOSITORY https://github.com/abseil/abseil-cpp.git
  GIT_TAG d9e4955c65cd4367dd6bf46f4ccb8cd3d100540b)
FetchContent_MakeAvailable(abseil)
set_property(DIRECTORY "${abseil_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
add_library(logitlab_core STATIC src/logistic.cpp)
add_library(logitlab::core ALIAS logitlab_core)
target_include_directories(logitlab_core PUBLIC include)
target_link_libraries(logitlab_core PUBLIC absl::span)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(logitlab_core PRIVATE -Wall -Wextra -Wpedantic)
  if(LOGITLAB_NATIVE)
    target_compile_options(logitlab_core PRIVATE -march=native)
  endif()
endif()
option(LOGITLAB_EXAMPLES "Build runnable C++ examples" ON)
if(LOGITLAB_EXAMPLES)
  add_executable(logitlab_train examples/train.cpp)
  target_link_libraries(logitlab_train PRIVATE logitlab::core)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(logitlab_tests tests/logitlab_test.cpp)
  target_link_libraries(logitlab_tests PRIVATE logitlab::core)
  add_test(NAME logitlab_tests COMMAND logitlab_tests)
  if(LOGITLAB_EXAMPLES)
    add_test(NAME logitlab_example COMMAND logitlab_train)
  endif()
endif()
if(LOGITLAB_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git
    GIT_TAG a2e59f0e7065404b44dfe92a28aca47ba1378dc4)
  FetchContent_MakeAvailable(pybind11)
  pybind11_add_module(_core bindings/python.cpp)
  target_link_libraries(_core PRIVATE logitlab::core)
  install(TARGETS _core LIBRARY DESTINATION logitlab)
endif()
if(LOGITLAB_LEGACY)
  add_library(legacy_data STATIC data.cpp)
  target_link_libraries(legacy_data PUBLIC absl::strings absl::flat_hash_map absl::flat_hash_set)
  foreach(name lr flrl flrl_predict sgd_lr lbfgs_lr pair_rank online_svrg)
    add_executable(legacy_${name} ${name}.cpp)
    target_link_libraries(legacy_${name} PRIVATE legacy_data)
  endforeach()
  target_link_libraries(legacy_lbfgs_lr PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lib/liblbfgs.a)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(legacy_lbfgs_lr PRIVATE -no-pie)
  endif()
endif()
